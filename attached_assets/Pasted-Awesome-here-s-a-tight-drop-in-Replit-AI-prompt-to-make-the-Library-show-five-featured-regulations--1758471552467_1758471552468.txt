If it doesn’t exist, create it per this SQL aggregation:

SELECT
  r.id,
  r.code,
  r.name_en AS "nameEn",
  r.name_ar AS "nameAr",
  r.version,
  r.status,
  COUNT(rc.id)::int AS "totalControls"
FROM regulations r
LEFT JOIN regulation_controls rc ON rc.regulation_id = r.id
GROUP BY r.id
ORDER BY r.code ASC;

Concrete snippet (React / Tailwind)

// client/src/pages/regulations/Library.tsx (or equivalent)
import React from 'react';

const FEATURED_CODES = [
  'NCA-ECC-2024',
  'DCC-2022',
  'CSCC-2023',
  'CRFR-2023',
  'MVC-2024',
];

function shortCode(code: string) {
  if (code.startsWith('NCA-ECC')) return 'ECC';
  if (code.startsWith('DCC')) return 'DCC';
  if (code.startsWith('CSCC')) return 'CSCC';
  if (code.startsWith('CRFR')) return 'CRFR';
  if (code.startsWith('MVC')) return 'MVC';
  return code;
}

function StatusPill({ status }: { status?: string }) {
  const label =
    status === 'active' ? 'Active' :
    status === 'coming_soon' ? 'Coming Soon' :
    'Draft';
  const cls =
    status === 'active' ? 'bg-emerald-100 text-emerald-700' :
    status === 'coming_soon' ? 'bg-amber-100 text-amber-700' :
    'bg-neutral-100 text-neutral-700';
  return <span className={`text-xs px-2 py-1 rounded-full ${cls}`}>{label}</span>;
}

export default function RegulationLibrary() {
  const [summary, setSummary] = React.useState<any[]>([]);
  React.useEffect(() => {
    (async () => {
      const r = await fetch('/api/regulations/summary');
      setSummary(await r.json());
    })();
  }, []);

  const byCode = React.useMemo(() => {
    const m = new Map<string, any>();
    for (const r of summary) m.set(r.code, r);
    return m;
  }, [summary]);

  const featured = FEATURED_CODES.map(code => {
    const r = byCode.get(code);
    if (!r) {
      console.warn('Featured regulation not found:', code);
      return { id: null, code, nameEn: '', status: 'coming_soon', totalControls: 0 };
    }
    return r;
  });

  return (
    <div className="p-6">
      <h1 className="text-2xl font-semibold">Regulation Library</h1>
      <p className="text-sm opacity-70">Explore and browse regulatory frameworks and controls</p>

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5 gap-4 mt-6">
        {featured.map((r) => {
          const label = shortCode(r.code);
          const title = `${label}${r.nameEn ? ` (${r.nameEn})` : ''}`;
          return (
            <div key={r.code} className="rounded-2xl border shadow-sm p-5 bg-white">
              <div className="flex items-start justify-between">
                <div className="text-lg font-semibold">{title}</div>
                <StatusPill status={r.status} />
              </div>

              {/* Optional description if your API provides it */}
              {r.description ? (
                <div className="text-sm opacity-70 mt-1 line-clamp-2">{r.description}</div>
              ) : null}

              <div className="mt-6 flex items-center justify-between">
                <div className="text-xs uppercase opacity-60">Total Controls</div>
                <div className="text-2xl font-bold">{r.totalControls ?? 0}</div>
              </div>

              <button
                className="mt-4 w-full border rounded-xl py-2 text-sm hover:bg-neutral-50 disabled:opacity-50"
                disabled={!r.id}
                onClick={() => { if (r.id) window.location.href = `/regulations/${r.id}`; }}
              >
                Open
              </button>
            </div>
          );
        })}
      </div>

      {/* Keep your existing Custom Regulations block below */}
      {/* ... */}
    </div>
  );
}
