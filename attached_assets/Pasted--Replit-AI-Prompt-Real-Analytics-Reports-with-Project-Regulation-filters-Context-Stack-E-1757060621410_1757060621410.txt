

Replit AI Prompt — “Real Analytics & Reports with Project/Regulation filters”

Context
	•	Stack: Express + TypeScript + Drizzle (Postgres), Vite/React/Tailwind + shadcn, React Query.
	•	Auth: req.user.id (string), req.user.claims.org (organization id).
	•	Goal: Compute true task counts across the whole system (org), and allow global filters:
	•	Filter by projects (multi-select)
	•	Filter by regulations (multi-select)
	•	The page must reflect exact counts (no fake data).

// tables: tasks, projects, users, task_regulations (junction), regulations
// Schema names may differ; adjust accordingly.

// base: eq(tasks.organizationId, org)
// if projectIds.length: inArray(tasks.projectId, projectIds)
// if regulationIds.length: inner join task_regulations on task.id = tr.taskId and inArray(tr.regulationId, regulationIds)

	•	Compute:
	•	totals.all: COUNT(*)

	•	totals.overdue: tasks with dueDate < now() and not completed
	•	totals.completedToday: completedAt::date = current_date

	•	byRegulation: join junction → GROUP BY regulationId
	•	byAssignee: GROUP BY assigneeId
	•	(Optional) bySeverity if you track severity on tasks.
	•	Return a single response object containing all aggregates.

1.2 Example (sketch code – adjust names)

// server/routes/analytics.ts
import { Router } from "express";
import { db } from "../db";
import { and, eq, inArray, sql, gt, lt } from "drizzle-orm";
// import your tables: tasks, projects, users, taskRegulations, regulations
// import zod for parsing if you prefer

const router = Router();

router.get("/tasks", async (req: any, res) => {
  const org = req.user?.claims?.org;
  if (!org) return res.status(400).json({ message: "Organization missing" });

  const projectIds = String(req.query.projectIds || "")
    .split(",").filter(Boolean).map((x) => Number(x));
  const regulationIds = String(req.query.regulationIds || "")
    .split(",").filter(Boolean);

  const dateFrom = req.query.dateFrom ? new Date(String(req.query.dateFrom)) : null;
  const dateTo = req.query.dateTo ? new Date(String(req.query.dateTo)) : null;

  // Build dynamic WHEREs
  const wheres: any[] = [eq(tasks.organizationId, org)];
  if (projectIds.length) wheres.push(inArray(tasks.projectId, projectIds));
  if (dateFrom) wheres.push(gt(tasks.createdAt, dateFrom));
  if (dateTo) wheres.push(lt(tasks.createdAt, dateTo));

  // If filtering by regulations, pre-resolve task IDs through the junction table
  let taskIdFilter: number[] | null = null;
  if (regulationIds.length) {
    const rows = await db.select({ taskId: taskRegulations.taskId })
      .from(taskRegulations)
      .innerJoin(tasks, and(
        eq(taskRegulations.taskId, tasks.id),
        eq(tasks.organizationId, org)
      ))
      .where(inArray(taskRegulations.regulationId, regulationIds));
    taskIdFilter = Array.from(new Set(rows.map(r => r.taskId)));
    if (taskIdFilter.length === 0) {
      return res.json({
        totals: { all: 0, byStatus: {}, overdue: 0, completedToday: 0 },
        byProject: [], byRegulation: [], byAssignee: [],
        dateRange: { from: req.query.dateFrom, to: req.query.dateTo }
      });
    }
    wheres.push(inArray(tasks.id, taskIdFilter));
  }

  const whereAll = and(...wheres);

  // totals.all
  const [{ count: all }] = await db.select({ count: sql<number>`COUNT(*)` })
    .from(tasks).where(whereAll);

  // totals.byStatus
  const byStatusRows = await db.select({
    status: tasks.status,
    count: sql<number>`COUNT(*)`
  }).from(tasks).where(whereAll).groupBy(tasks.status);

  // overdue (dueDate past, not done)
  const overdueRows = await db.select({ count: sql<number>`COUNT(*)` })
    .from(tasks)
    .where(and(
      whereAll,
      sql`COALESCE(${tasks.dueDate}, 'infinity') < now()`,
      sql`${tasks.status} <> 'completed'`
    ));
  const overdue = overdueRows[0]?.count ?? 0;

  // completedToday
  const completedTodayRows = await db.select({ count: sql<number>`COUNT(*)` })
    .from(tasks)
    .where(and(
      whereAll,
      sql`${tasks.status} = 'completed'`,
      sql`DATE(${tasks.completedAt}) = CURRENT_DATE`
    ));
  const completedToday = completedTodayRows[0]?.count ?? 0;

  // byProject
  const byProject = await db.select({
    projectId: tasks.projectId,
    count: sql<number>`COUNT(*)`,
  }).from(tasks).where(whereAll).groupBy(tasks.projectId);

  // byRegulation
  const byRegulation = await db.select({
    regulationId: taskRegulations.regulationId,
    count: sql<number>`COUNT(*)`,
  })
  .from(taskRegulations)
  .innerJoin(tasks, and(eq(taskRegulations.taskId, tasks.id), whereAll))
  .groupBy(taskRegulations.regulationId);

  // byAssignee
  const byAssignee = await db.select({
    assigneeId: tasks.assigneeId,
    count: sql<number>`COUNT(*)`,
  }).from(tasks).where(whereAll).groupBy(tasks.assigneeId);

  // Optionally join names for project/regulation/assignee:
  // (left joins or follow-up queries; keep it simple if you prefer)

  res.json({
    totals: {
      all: Number(all) || 0,
      byStatus: Object.fromEntries(byStatusRows.map(r => [r.status, Number(r.count)])),
      overdue: Number(overdue) || 0,
      completedToday: Number(completedToday) || 0,
    },
    byProject: byProject.map(r => ({ projectId: r.projectId, name: null, count: Number(r.count) })),
    byRegulation: byRegulation.map(r => ({ regulationId: r.regulationId, code: null, count: Number(r.count) })),
    byAssignee: byAssignee.map(r => ({ userId: String(r.assigneeId || ""), name: null, count: Number(r.count) })),
    dateRange: { from: req.query.dateFrom, to: req.query.dateTo }
  });
});

export default router;

2.1 Data hook

// client/src/lib/api/analytics.ts
export type AnalyticsResponse = {
  totals: { all: number; byStatus: Record<string, number>; overdue: number; completedToday: number; };
  byProject: Array<{ projectId: number; name: string | null; count: number }>;
  byRegulation: Array<{ regulationId: string | number; code: string | null; count: number }>;
  byAssignee: Array<{ userId: string; name: string | null; count: number }>;
  dateRange?: { from?: string; to?: string };
};

export async function fetchTaskAnalytics(params: {
  projectIds?: number[];
  regulationIds?: (string | number)[];
  dateFrom?: string;
  dateTo?: string;
}) {
  const qs = new URLSearchParams();
  if (params.projectIds?.length) qs.set("projectIds", params.projectIds.join(","));
  if (params.regulationIds?.length) qs.set("regulationIds", params.regulationIds.join(","));
  if (params.dateFrom) qs.set("dateFrom", params.dateFrom);
  if (params.dateTo) qs.set("dateTo", params.dateTo);

  const res = await fetch(`/api/analytics/tasks?${qs.toString()}`, { credentials: "include" });
  if (!res.ok) throw new Error(await res.text());
  return res.json() as Promise<AnalyticsResponse>;
}

// client/src/pages/AnalyticsReports.tsx
import { useQuery } from "@tanstack/react-query";
import { fetchTaskAnalytics } from "@/lib/api/analytics";
import { useState } from "react";
import { Card, CardHeader, CardTitle, CardContent } from "@/components/ui/card";
import { BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer } from "recharts";
import MultiProjectFilter from "@/components/filters/MultiProjectFilter";
import MultiRegulationFilter from "@/components/filters/MultiRegulationFilter";

export default function AnalyticsReports() {
  const [projectIds, setProjectIds] = useState<number[]>([]);
  const [regulationIds, setRegulationIds] = useState<(string|number)[]>([]);

  const { data, isLoading } = useQuery({
    queryKey: ["analytics-tasks", { projectIds, regulationIds }],
    queryFn: () => fetchTaskAnalytics({ projectIds, regulationIds }),
    staleTime: 30_000,
  });

  const totals = data?.totals;
  const byStatus = Object.entries(totals?.byStatus || {}).map(([k,v]) => ({ name: k, value: v }));
  const byProject = (data?.byProject || []).map(r => ({ name: String(r.projectId), value: r.count }));
  const byRegulation = (data?.byRegulation || []).map(r => ({ name: String(r.regulationId), value: r.count }));
  const byAssignee = (data?.byAssignee || []).map(r => ({ name: r.name || r.userId, value: r.count }));

  return (
    <div className="space-y-6">
      {/* Filters */}
      <div className="flex flex-wrap gap-3">
        <MultiProjectFilter value={projectIds} onChange={setProjectIds} />
        <MultiRegulationFilter value={regulationIds} onChange={setRegulationIds} />
      </div>

      {/* KPIs */}
      <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
        <Card><CardHeader><CardTitle>Total Tasks</CardTitle></CardHeader><CardContent className="text-3xl font-semibold">{totals?.all ?? 0}</CardContent></Card>
        <Card><CardHeader><CardTitle>Overdue</CardTitle></CardHeader><CardContent className="text-3xl font-semibold">{totals?.overdue ?? 0}</CardContent></Card>
        <Card><CardHeader><CardTitle>Completed Today</CardTitle></CardHeader><CardContent className="text-3xl font-semibold">{totals?.completedToday ?? 0}</CardContent></Card>
        <Card><CardHeader><CardTitle>Statuses</CardTitle></CardHeader><CardContent>{byStatus.reduce((a,c)=>a+c.value,0)} total</CardContent></Card>
      </div>

      {/* Charts */}
      <div className="grid grid-cols-1 xl:grid-cols-2 gap-6">
        <Card>
          <CardHeader><CardTitle>Tasks by Status</CardTitle></CardHeader>
          <CardContent className="h-72">
            <ResponsiveContainer width="100%" height="100%">
              <BarChart data={byStatus}><XAxis dataKey="name" /><YAxis allowDecimals={false} /><Tooltip /><Bar dataKey="value" /></BarChart>
            </ResponsiveContainer>
          </CardContent>
        </Card>

        <Card>
          <CardHeader><CardTitle>Tasks by Project</CardTitle></CardHeader>
          <CardContent className="h-72">
            <ResponsiveContainer width="100%" height="100%">
              <BarChart data={byProject}><XAxis dataKey="name" /><YAxis allowDecimals={false} /><Tooltip /><Bar dataKey="value" /></BarChart>
            </ResponsiveContainer>
          </CardContent>
        </Card>

        <Card>
          <CardHeader><CardTitle>Tasks by Regulation</CardTitle></CardHeader>
          <CardContent className="h-72">
            <ResponsiveContainer width="100%" height="100%">
              <BarChart data={byRegulation}><XAxis dataKey="name" /><YAxis allowDecimals={false} /><Tooltip /><Bar dataKey="value" /></BarChart>
            </ResponsiveContainer>
          </CardContent>
        </Card>

        <Card>
          <CardHeader><CardTitle>Tasks by Assignee</CardTitle></CardHeader>
          <CardContent className="h-72">
            <ResponsiveContainer width="100%" height="100%">
              <BarChart data={byAssignee}><XAxis dataKey="name" hide={byAssignee.length>12} /><YAxis allowDecimals={false} /><Tooltip /><Bar dataKey="value" /></BarChart>
            </ResponsiveContainer>

        </Card>
      </div>
    </div>
  );
}
