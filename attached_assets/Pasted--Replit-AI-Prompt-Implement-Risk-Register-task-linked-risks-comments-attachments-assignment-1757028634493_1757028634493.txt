

Replit AI Prompt — Implement “Risk Register” (task-linked risks, comments, attachments, assignment)

Context (stack + conventions)
	•	Frontend: Vite + React + Tailwind + shadcn/Radix, React Hook Form, React Query.
	•	Backend: Express (TypeScript), Drizzle ORM. Auth via req.user with id (string) and claims.org (organization).
	•	Existing features to reuse: tasks table & routes, unified comments (used by tasks/projects), attachments/uploads service, email service, WS updates if available.

Goal

Add a Risk Register with these behaviors:
	1.	Task → Risk toggle: any task can be marked “Risk” (toggle on the task).
	2.	Risk item is created and linked to the original task; shown in a dedicated Risk Register screen.
	3.	Risk fields (editable by compliance officer):
	•	Risk Description
	•	Risk Status: Not started, In Progress, Mitigated, Under Review, Completed
	•	Risk Severity: Low, Medium, High, Critical, Urgent
	•	Mitigation Plan Description
	•	Assignee (any collaborator in the org)
	•	Attachments (upload files)
	•	Comments (same component/behavior as tasks)
	4.	Toggle off risk: when mitigated/resolved, officer can “Remove risk” which closes the risk & clears the flag on the task.
	5.	Permissions:
	•	View: users who can view the original task.
	•	Edit risk fields / assign / close risk: compliance officer role or admins; task creator can view & comment.
	6.	Notifications: when a risk is created or assigned, email the assignee and watchers (reuse email service). (If email envs are missing, fail gracefully.)

Create shared/risk.ts with a new table risks (one risk row per task when toggled):

// shared/risk.ts
import {
  pgTable, serial, integer, varchar, text, timestamp, boolean
} from "drizzle-orm/pg-core";

export const riskStatus = ["not-started","in-progress","mitigated","under-review","completed"] as const;
export type RiskStatus = typeof riskStatus[number];

export const riskSeverity = ["low","medium","high","critical","urgent"] as const;
export type RiskSeverity = typeof riskSeverity[number];

export const risks = pgTable("risks", {
  id: serial("id").primaryKey(),

  organizationId: varchar("organization_id", { length: 64 }).notNull(),

  // Link to the original task
  taskId: integer("task_id").notNull(),

  // Assignment (string user id)
  assigneeId: varchar("assignee_id", { length: 128 }), // nullable

  // Core fields
  title: varchar("title", { length: 256 }).notNull(), // copy from task title at creation
  riskDescription: text("risk_description"),
  mitigationPlan: text("mitigation_plan"),

  status: varchar("status", { length: 32 }).$type<RiskStatus>().default("not-started").notNull(),
  severity: varchar("severity", { length: 32 }).$type<RiskSeverity>().default("medium").notNull(),

  // Lifecycle
  isOpen: boolean("is_open").default(true).notNull(),
  createdById: varchar("created_by_id", { length: 128 }).notNull(),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),
  closedAt: timestamp("closed_at", { withTimezone: true }),
});

Routes

import { Router } from "express";
import { z } from "zod";
import { db } from "../db";
import { risks, riskStatus, riskSeverity } from "@shared/risk";
import { tasks } from "@shared/schema"; // your tasks table
import { and, eq, desc } from "drizzle-orm";
import emailService from "../emailService"; // unified service

const router = Router();

// GET /api/risks?status=&severity=&assigneeId=&q=&limit=&cursor=
router.get("/", async (req:any, res) => {
  if (!req.user?.claims?.org) return res.status(401).end();
  const org = req.user.claims.org;

  const schema = z.object({
    status: z.enum(riskStatus).optional(),
    severity: z.enum(riskSeverity).optional(),
    assigneeId: z.string().optional(),
    q: z.string().optional(),
    limit: z.coerce.number().min(1).max(100).default(50),
    cursor: z.coerce.number().optional(), // risk.id for pagination
  });
  const q = schema.parse(req.query);

  // basic where
  let where: any = and(eq(risks.organizationId, org));
  // add filters if provided (status/severity/assignee)
  if (q.status) where = and(where, eq(risks.status, q.status));
  if (q.severity) where = and(where, eq(risks.severity, q.severity));
  if (q.assigneeId) where = and(where, eq(risks.assigneeId, q.assigneeId));
  if (q.cursor) where = and(where, desc(risks.id).lt(q.cursor as any));

  const rows = await db.select().from(risks)
    .where(where)
    .orderBy(desc(risks.id))
    .limit(q.limit);

  res.json({ items: rows, nextCursor: rows.length ? rows[rows.length - 1].id : null });
});

// POST /api/risks/toggle  { taskId, makeRisk:boolean }
router.post("/toggle", async (req:any, res) => {
  if (!req.user?.claims?.org) return res.status(401).end();
  const org = req.user.claims.org;

  const body = z.object({
    taskId: z.number(),
    makeRisk: z.boolean(),
  }).parse(req.body);

  // fetch task (ensure org & view permissions)
  const [task] = await db.select().from(tasks).where(and(eq(tasks.id, body.taskId), eq(tasks.organizationId, org)));
  if (!task) return res.status(404).json({ message: "Task not found" });
  // TODO: canViewTask check if you have one

  if (body.makeRisk) {
    // set task.is_risk true
    await db.update(tasks).set({ isRisk: true, updatedAt: new Date() }).where(eq(tasks.id, task.id));

    // create risk if not exists for this task & still open
    const existing = await db.select().from(risks)
      .where(and(eq(risks.organizationId, org), eq(risks.taskId, task.id), eq(risks.isOpen, true)));
    if (existing.length) return res.json(existing[0]);

    const [risk] = await db.insert(risks).values({
      organizationId: org,
      taskId: task.id,
      title: task.title,
      createdById: req.user.id,
      status: "not-started",
      severity: "medium",
      isOpen: true,
    }).returning();

    // Optional: notify task assignee/owner
    try {
      if (task.assigneeEmail) {
        await emailService.sendEmailWithRetry({
          to: task.assigneeEmail,
          subject: `[Ambersand] Task marked as Risk: ${task.title}`,
          html: `<p>The task <strong>${task.title}</strong> has been marked as a risk.</p>`,
        });
      }
    } catch {}

    return res.status(201).json(risk);
  } else {
    // turning off: close open risk(s) & set task.is_risk false
    await db.update(tasks).set({ isRisk: false, updatedAt: new Date() }).where(eq(tasks.id, task.id));
    const [closed] = await db.update(risks).set({
      isOpen: false, closedAt: new Date(), status: "mitigated", updatedAt: new Date(),
    }).where(and(eq(risks.organizationId, org), eq(risks.taskId, task.id), eq(risks.isOpen, true)))
      .returning();
    return res.json(closed || { ok: true });
  }
});

// PATCH /api/risks/:id  (edit risk fields) — compliance officer/admin
router.patch("/:id", async (req:any, res) => {
  if (!req.user?.claims?.org) return res.status(401).end();
  const org = req.user.claims.org;

  const id = Number(req.params.id);
  const schema = z.object({
    title: z.string().min(1).max(256).optional(),
    riskDescription: z.string().optional(),
    mitigationPlan: z.string().optional(),
    status: z.enum(riskStatus).optional(),
    severity: z.enum(riskSeverity).optional(),
    assigneeId: z.string().optional().nullable(),
    isOpen: z.boolean().optional(),
  });
  const body = schema.parse(req.body);

  // permission: only compliance officer/admin
  if (!(req.user?.isAdmin || req.user?.roles?.includes?.("compliance-officer"))) {
    return res.status(403).end();
  }

  const [updated] = await db.update(risks).set({
    ...body,
    updatedAt: new Date(),
  }).where(and(eq(risks.id, id), eq(risks.organizationId, org)))
    .returning();

  if (!updated) return res.status(404).end();
  res.json(updated);
});

export default router;

Mount the router in your server/routes.ts:

5) Client hooks & types

Create client/src/lib/api/risk.ts:

export type RiskStatus = "not-started"|"in-progress"|"mitigated"|"under-review"|"completed";
export type RiskSeverity = "low"|"medium"|"high"|"critical"|"urgent";

export interface RiskItem {
  id: number;
  organizationId: string;
  taskId: number;
  assigneeId?: string | null;
  title: string;
  riskDescription?: string | null;
  mitigationPlan?: string | null;
  status: RiskStatus;
  severity: RiskSeverity;
  isOpen: boolean;
  createdById: string;
  createdAt: string;
  updatedAt: string;
  closedAt?: string | null;
}

// Queries
export async function listRisks(params: URLSearchParams) {
  const res = await fetch(`/api/risks?${params.toString()}`, { credentials: "include" });
  if (!res.ok) throw new Error("Failed to list risks");
  return res.json();
}

export async function toggleRisk(taskId:number, makeRisk:boolean) {
  const res = await fetch(`/api/risks/toggle`, {
    method: "POST", credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ taskId, makeRisk })
  });
  if (!res.ok) throw new Error("Failed to toggle risk");
  return res.json();
}

export async function updateRisk(id:number, patch: Partial<RiskItem>) {
  const res = await fetch(`/api/risks/${id}`, {
    method: "PATCH", credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(patch),
  });
  if (!res.ok) throw new Error("Failed to update risk");
  return res.json();
}
