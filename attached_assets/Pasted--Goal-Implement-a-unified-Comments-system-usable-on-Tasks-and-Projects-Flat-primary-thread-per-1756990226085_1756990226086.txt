

Goal

Implement a unified Comments system usable on Tasks and Projects:
	•	Flat primary thread per target (Task or Project), with optional short replies (1 level deep).
	•	Mentions (@name) with auto-complete from org users; notify mentioned users.
	•	Real-time updates via WebSocket broadcast.
	•	Email notifications (optional toggle) for mentions and for subscribers (assignee, owner).
	•	Attachments (optional later): keep DB-ready now, UI can come later.
	•	Permissions: only users who can view the target can read; only users who can edit/comment the target can post/edit/delete their own comments; admins can delete.

import { pgTable, serial, integer, text, varchar, timestamp, boolean } from "drizzle-orm/pg-core";
import { sql } from "drizzle-orm";

export const commentTargets = ["task","project"] as const;
export type CommentTarget = typeof commentTargets[number];

export const comments = pgTable("comments", {
  id: serial("id").primaryKey(),
  organizationId: varchar("organization_id", { length: 64 }).notNull(),
  targetType: varchar("target_type", { length: 16 }).$type<CommentTarget>().notNull(), // 'task' | 'project'
  targetId: integer("target_id").notNull(),
  parentId: integer("parent_id"), // for replies (1-level)
  authorId: integer("author_id").notNull(),
  body: text("body").notNull(),
  mentions: text("mentions").default(sql`'[]'::text`).notNull(), // JSON stringified array of userIds
  hasAttachments: boolean("has_attachments").default(false).notNull(),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).defaultNow().notNull(),

});

Routes

// server/routes/comments.ts
import { Router } from "express";
import { z } from "zod";
import { db } from "../db";
import { comments, commentSubscriptions } from "@shared/comments";
import { isAuthenticated } from "../replitAuth";
import { and, eq, desc, isNull } from "drizzle-orm";
import { parseMentions } from "../services/mentions"; // implement below
import { notifyComment } from "../services/notifications"; // implement below

const router = Router();

// GET list (paginated)
router.get("/", isAuthenticated, async (req:any, res) => {
  const schema = z.object({
    targetType: z.enum(["task","project"]),
    targetId: z.coerce.number(),
    cursor: z.coerce.number().optional(), // comment id before which to load
    limit: z.coerce.number().min(1).max(100).default(50),
  });
  const q = schema.parse(req.query);
  // TODO: canViewTarget check
  const whereBase = and(
    eq(comments.organizationId, req.user.claims.org),
    eq(comments.targetType, q.targetType),
    eq(comments.targetId, q.targetId),
    isNull(comments.deletedAt)
  );
  const where = q.cursor ? and(whereBase, desc(comments.id).lt(q.cursor as any)) : whereBase;
  const rows = await db.select().from(comments)
    .where(where as any)
    .orderBy(desc(comments.id))
    .limit(q.limit);
  res.json({ items: rows, nextCursor: rows.length ? rows[rows.length-1].id : null });
});

// POST create
router.post("/", isAuthenticated, async (req:any, res) => {
  const schema = z.object({
    targetType: z.enum(["task","project"]),
    targetId: z.number(),
    parentId: z.number().optional(),
    body: z.string().min(1).max(4000),
  });
  const body = schema.parse(req.body);
  // TODO: canCommentTarget check
  const mentions = await parseMentions(body.body, req.user.claims.org);

  const [row] = await db.insert(comments).values({
    organizationId: req.user.claims.org,
    targetType: body.targetType,
    targetId: body.targetId,
    parentId: body.parentId ?? null,
    authorId: req.user.id,
    body: body.body,
    mentions: JSON.stringify(mentions.userIds),
  }).returning();

  // auto-subscribe author; also ensure assignee/owner subscribed elsewhere
  await db.insert(commentSubscriptions).values({
    organizationId: req.user.claims.org,
    targetType: body.targetType,
    targetId: body.targetId,
    userId: req.user.id,
  }).onConflictDoNothing();

  // notify via ws + email
  await notifyComment({ comment: row, mentions });

  res.status(201).json(row);
});

// PATCH edit (author only)
router.patch("/:id", isAuthenticated, async (req:any, res) => {
  const schema = z.object({ body: z.string().min(1).max(4000) });
  const { body } = schema.parse(req.body);
  const id = Number(req.params.id);
  const [existing] = await db.select().from(comments)
    .where(and(eq(comments.id, id), eq(comments.organizationId, req.user.claims.org)));
  if (!existing) return res.sendStatus(404);
  if (existing.authorId !== req.user.id && !req.user.isAdmin) return res.sendStatus(403);

  const mentions = await parseMentions(body, req.user.claims.org);
  const [row] = await db.update(comments).set({
    body, mentions: JSON.stringify(mentions.userIds),
    updatedAt: new Date(),
  }).where(eq(comments.id, id)).returning();

  await notifyComment({ comment: row, mentions, edited: true });
  res.json(row);
});

// DELETE (soft delete; author or admin)
router.delete("/:id", isAuthenticated, async (req:any, res) => {
  const id = Number(req.params.id);
  const [existing] = await db.select().from(comments)
    .where(and(eq(comments.id, id), eq(comments.organizationId, req.user.claims.org)));
  if (!existing) return res.sendStatus(404);
  if (existing.authorId !== req.user.id && !req.user.isAdmin) return res.sendStatus(403);

  await db.update(comments).set({ deletedAt: new Date() }).where(eq(comments.id, id));
  res.sendStatus(204);
});

export default router;

// parse @mentions in the body -> return { userIds: number[], tokens: Array<{id:number, name:string}> }
export async function parseMentions(body:string, orgId:string) {
  // Regex for @mentions (alnum, dots, underscores). Replace with your user directory lookup.
  const handles = Array.from(new Set((body.match(/@[\w.\-]+/g) || []).map(s => s.slice(1))));
  // Lookup users by handle/name within orgId
  const users = await findUsersByHandles(handles, orgId); // implement according to your users table
  return { userIds: users.map(u => u.id), users };
}

4) WebSocket (real-time)

Use your existing ws server. Create a small hub:

server/ws/comments.ts

import { wss } from "./wss"; // your existing ws server singleton
import type { WebSocket } from "ws";

export function roomFor(orgId:string, targetType:"task"|"project", targetId:number) {
  return `${orgId}:${targetType}:${targetId}`;
}

export async function broadcastComment(c:any) {
  const room = roomFor(c.organizationId, c.targetType, c.targetId);
  wss.clients.forEach((client:WebSocket & { room?:string }) => {
    if (client.readyState === 1 && client.room === room) {
      client.send(JSON.stringify({ type: "comment:new", payload: c }));
    }
  });
}
